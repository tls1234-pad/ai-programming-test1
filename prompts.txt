# load_scores.py
# ---------------------------------------------------------------------------

#
# 명세 (10주차 명세서 6칸 요약)
#   load_scores(path) -> (kept, skipped)
#     kept    : [(이름, 점수), ...]      점수는 0~100 사이의 정수
#     skipped : [(줄번호, 원본줄, 이유), ...]   건너뛴 줄의 기록
#
#   ① 파일이 없으면        → FileNotFoundError 를 그대로 올립니다 (숨기지 않습니다)
#   ② 헤더만 있으면        → ([], []) 를 돌려줍니다
#   ③ 열 개수가 2가 아니면 → 건너뛰고 기록합니다
#   ④ 점수가 비었거나 정수가 아니면 → 건너뛰고 기록합니다
#   ⑤ 점수가 0~100 밖이면  → 건너뛰고 기록합니다
#   ⑥ 이름이 비었으면      → 건너뛰고 기록합니다
#
#   ★ 같은 이름이 두 번 나오는 경우 (v5 재검토에서 바꾼 부분)
#     이 함수는 지우지 않고 둘 다 남깁니다.
#     이름이 같다고 같은 사람이라는 보장이 없기 때문입니다(동명이인).
#     누구를 남길지는 이 함수가 아니라 쓰는 쪽에서 정할 일입니다.
#
#   빈 줄은 기록 없이 그냥 건너뜁니다.
# ---------------------------------------------------------------------------

import csv

MIN_SCORE = 0
MAX_SCORE = 100


def load_scores(path):
    """성적 CSV를 읽어 (유효한 목록, 건너뛴 줄 기록) 을 돌려준다."""
    kept = []
    skipped = []

    with open(path, "r", encoding="utf-8-sig", newline="") as f:
        reader = csv.reader(f)
        for line_no, row in enumerate(reader, start=1):
            if line_no == 1:              # 헤더
                continue
            if not row:                   # 빈 줄 — 기록하지 않는다
                continue

            원본 = ",".join(row)

            if len(row) != 2:
                skipped.append((line_no, 원본, "열 개수가 2가 아님"))
                continue

            name = row[0].strip()
            raw = row[1].strip()

            if name == "":
                skipped.append((line_no, 원본, "이름이 비어 있음"))
                continue
            if raw == "":
                skipped.append((line_no, 원본, "점수가 비어 있음"))
                continue

            try:
                score = int(raw)
            except ValueError:
                skipped.append((line_no, 원본, "점수가 정수가 아님"))
                continue

            if score < MIN_SCORE or score > MAX_SCORE:
                skipped.append((line_no, 원본, "점수가 0~100 밖"))
                continue

            kept.append((name, score))

    return kept, skipped


if __name__ == "__main__":
    import sys

    path = sys.argv[1] if len(sys.argv) > 1 else "scores.csv"
    kept, skipped = load_scores(path)
    print(f"읽은 줄 : {len(kept)}개")
    for name, score in kept:
        print(f"  {name} {score}")
    print(f"건너뛴 줄 : {len(skipped)}개")
    for line_no, 원본, 이유 in skipped:
        print(f"  {line_no}행  {원본}   ← {이유}")

── 붙여 넣을 명세서 6칸 (10주차에 직접 채운 것을 쓰세요) ──
  ① 무엇을 받나       :
    - load_scores(path) 함수의 입력으로 성적 CSV 파일의 경로를 받는다.
    - CSV 헤더는 name,score이고, 데이터는 이름과 점수의 두 열이다.
    - UTF-8 BOM을 허용한다.
    - CSV 규칙에 따라 따옴표 안의 쉼표를 하나의 이름에 포함된 문자로 처리한다.

  ② 무엇을 돌려주나   :
    - (유효한 성적 목록, 제외한 행의 기록 목록)을 반환한다.
    - 유효한 성적은 (이름, 정수 점수) 쌍이며 원래 입력 순서를 유지한다.
    - 제외한 행의 기록에는 줄 번호와 제외 사유가 있어야 한다.
    - 헤더만 있는 파일은 ([], [])를 반환한다.
    - 제외 기록 한 항목의 구체적인 자료형과 필드 순서는 기존 명세에 명시되어 있지 않다.

  ③ 실패하면 어떻게 하나 :
    - 파일이 없거나 읽을 수 없으면 파일 오류를 호출자에게 전달한다.
    - 파일 접근 실패를 정상적인 빈 결과로 숨기지 않는다.
    - 헤더가 없거나 올바르지 않으면 헤더 오류로 처리한다.
    - 데이터 행의 열 수, 이름, 점수 등이 규칙을 어기면 그 행을 제외하고 줄 번호와 사유를 기록한 뒤 다음 행을 계속 읽는다.
    - 헤더 오류의 구체적인 예외 클래스와 예외 메시지는 기존 명세에 명시되어 있지 않다.

  ④ 쓰면 안 되는 것   :
    - load_scores 함수 구현에는 외부 라이브러리를 사용하지 않고 표준 라이브러리만 사용한다. 이는 테스트 프레임워크 pytest를 금지한다는 뜻은 아니다.
    - 원본 파일을 수정하거나 쓰기 모드로 열지 않는다.
    - 빈 점수를 0점으로 바꾸지 않는다.
    - 범위를 벗어난 점수나 잘못된 점수 표기를 유효한 성적으로 포함하지 않는다.
    - 이미 유효한 성적으로 채택한 이름의 뒤쪽 중복 성적을 또 포함하지 않는다.

  ⑤ 예외 상황         :
    - 존재하지 않는 파일과 헤더만 있는 파일은 서로 다른 상황으로 처리한다.
    - 첫 줄 자체가 없는 완전히 빈 파일은 헤더 오류로 처리한다.
    - 열 개수가 두 개가 아닌 데이터 행은 제외한다.
    - 양끝 공백을 제거했을 때 이름 또는 점수가 비어 있으면 제외한다.
    - 88.5 같은 소수나 '7 2'처럼 숫자 사이에 공백이 있는 점수는 제외한다.
    - '  77'처럼 점수 양끝에만 공백이 있으면 77점으로 인정한다.
    - 따옴표로 묶인 '최,네번'은 한 사람의 이름으로 보존한다.
    - 동일한 이름의 유효한 성적이 88점, 91점 순서로 나오면 88점만 남기고 91점 행은 중복으로 기록한다.
    - 명세의 중복 이름 판단은 이름 양끝의 공백을 제거한 뒤 수행한다.

  ⑥ 조건 · 제약       :
    - 점수는 양끝 공백을 뺀 십진 정수 0~100만 인정한다.
    - 경계값 0과 100은 유효하며, -1과 101은 유효하지 않다.
    - 이름과 점수 양끝의 공백을 제거한다.
    - 같은 이름이 반복되면 첫 번째 유효 성적만 남긴다.
    - 파일은 읽기 모드로 연다.
    - 함수 구현은 주석과 빈 줄을 제외하고 함수당 되도록 20줄 이내로 작성한다는 작성 제약이 있다.
    - 최종 재검토 조건: UTF-8 BOM 처리 후 헤더 원문이 name,score인 경우만 받는다. 따옴표로 감싼 헤더는 받지 않는다.
    - 최종 재검토 조건: 올바른 CSV 인용 규칙으로 여러 줄에 걸친 긴 이름은 전체를 한 사람으로 읽어야 한다. 그 안에 들어 있는 '가짜,99'를 별도 학생으로 만들면 안 된다.
    - 최종 재검토 조건: 인용하지 않은 이름 중간에 잘못된 따옴표가 있으면 그 행을 기록하고 제외한다.
    - 순수한 빈 데이터 줄을 기록할지, 제외 사유의 정확한 문구, 여러 줄 CSV 레코드의 줄 번호 기준 등은 기존 명세에서 명확하게 정하지 않았다.
